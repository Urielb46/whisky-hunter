3. `packages/scraper/src/queue/scheduler.ts`:
```ts
import type { Queue } from 'bullmq';

export interface RetailerScheduleConfig {
  id: string;
  cronExpression: string;
  catalogUrl: string;
  active: boolean;
}

/**
 * Idempotently register one BullMQ upsertJobScheduler per active retailer.
 * Safe to call on every worker startup (RESEARCH.md Pattern 1).
 * DATA-04: Each retailer scrapes on its own cron schedule.
 */
export async function registerRetailerSchedules(
  queue: Queue,
  retailers: RetailerScheduleConfig[],
): Promise<number> {
  let registered = 0;
  for (const retailer of retailers) {
    if (!retailer.active) continue;
    await queue.upsertJobScheduler(
      `scrape-${retailer.id}`,
      { pattern: retailer.cronExpression },
      {
        name: 'scrape-retailer',
        data: { retailerId: retailer.id, url: retailer.catalogUrl },
        opts: {
          attempts: 3,
          backoff: { type: 'exponential', delay: 30_000 },
          removeOnComplete: { count: 50 },
          removeOnFail: { count: 100 },
        },
      },
    );
    registered++;
  }
  return registered;
}
```

import { chromium } from 'playwright-extra';
import StealthPlugin from 'puppeteer-extra-plugin-stealth';

// Apply stealth plugin once at module load (idempotent)
chromium.use(StealthPlugin());

export interface StealthContext {
  browser: Awaited<ReturnType<typeof chromium.launch>>;
  ctx: Awaited<ReturnType<typeof import('playwright').BrowserType.prototype.launch>> extends infer B
    ? never
    : never; // typed as playwright.BrowserContext below via module augment
}

export async function createStealthContext(proxyUrl?: string) {
  const browser = await chromium.launch({
    headless: process.env.SCRAPER_HEADLESS !== 'false',
    args: ['--no-sandbox', '--disable-setuid-sandbox'],
  });
  const ctx = await browser.newContext({
    proxy: proxyUrl ? { server: proxyUrl } : undefined,
    viewport: { width: 1366, height: 768 },
    userAgent:
      'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36',
  });
  // Block resources that scrapers don't need — saves bandwidth and avoids fingerprinting by ad networks
  await ctx.route('**/*.{png,jpg,jpeg,gif,webp,svg,woff,woff2,ttf,otf,css}', (r) => r.abort());
  return { browser, ctx };
}

function makeMockQueue() {

  };

const RETAILERS: RetailerScheduleConfig[] = [
  { id: 'whisky-exchange', cronExpression: '0 2 * * *', catalogUrl: 'https://example.com/wex', active: true },
  { id: 'master-of-malt',  cronExpression: '30 2 * * *', catalogUrl: 'https://example.com/mom', active: true },
  { id: 'inactive-shop',   cronExpression: '0 3 * * *', catalogUrl: 'https://example.com/inactive', active: false },

describe('registerRetailerSchedules', () => {
  it('calls upsertJobScheduler once per active retailer', async () => {
    const queue = makeMockQueue();
    const count = await registerRetailerSchedules(queue as any, RETAILERS);
    expect(queue.upsertJobScheduler).toHaveBeenCalledTimes(2); // inactive skipped
    expect(count).toBe(2);
  });

  it('uses retailer id as scheduler key prefix', async () => {
    const queue = makeMockQueue();
    await registerRetailerSchedules(queue as any, RETAILERS);
    const calls = queue.upsertJobScheduler.mock.calls;
    expect(calls[0]![0]).toBe('scrape-whisky-exchange');
    expect(calls[1]![0]).toBe('scrape-master-of-malt');

    const queue = makeMockQueue();
    await registerRetailerSchedules(queue as any, RETAILERS);
    const registeredIds = queue.upsertJobScheduler.mock.calls.map((c: any[]) => c[0] as string);
    expect(registeredIds).not.toContain('scrape-inactive-shop');

  it('is idempotent — second call re-registers same schedulers without error', async () => {

import type { Page } from 'playwright';
import type { ScraperAdapter, RawProductScrape } from './types';

export const whiskyExchangeAdapter: ScraperAdapter = {
  retailerId: 'whisky-exchange',

  async fetchCatalogPage(page, pageNum) {
    await page.goto(
      `https://www.thewhiskyexchange.com/c/40/single-malt-scotch-whisky?pg=${pageNum}`,
      { waitUntil: 'domcontentloaded', timeout: 30_000 },
    );
    const products = await page.$$eval('.product-card', (cards) =>
      cards.map((card) => ({
        sourceProductId: card.querySelector('a')?.getAttribute('href')?.split('/').at(-1) ?? '',
        name: (card.querySelector('.product-card__name') as HTMLElement | null)?.innerText?.trim() ?? '',
        priceLocal: parseFloat(
          ((card.querySelector('.product-card__price') as HTMLElement | null)?.innerText ?? '0')
            .replace(/[^\d.]/g, ''),
        ),
        currency: 'GBP' as const,
        inStock: !card.classList.contains('product-card--unavailable'),
        url: `https://www.thewhiskyexchange.com${(card.querySelector('a') as HTMLAnchorElement | null)?.getAttribute('href') ?? ''}`,
        imageUrl: (card.querySelector('img') as HTMLImageElement | null)?.src,
      })),
    );
    // Pitfall 2: assert products exist on page 1 — empty array = Cloudflare challenge
    if (pageNum === 1 && products.length === 0) {
      throw new Error('Zero products on page 1 — possible bot block [whisky-exchange]');
    }
    return products as RawProductScrape[];
  },

  async totalPages(page) {
    try {
      const text = await page.$eval('.pagination__total', (el) => el.textContent ?? '1');
      return parseInt(text, 10) || 1;
    } catch {
      return 1;
    }
  },
};

// DATA-04: Record health metrics after every scrape job
export function registerHealthListeners(): QueueEvents {
  const events = new QueueEvents('scrape', { connection: redis });

  events.on('completed', async ({ jobId }) => {
    const job = await scrapeQueue.getJob(jobId);
    if (!job) return;
    const { retailerId } = job.data as { retailerId: string };
    await db
      .insert(scraperHealth)
      .values({
        retailerId,
        lastScrapedAt: new Date(),
        lastSuccessfulScrapeAt: new Date(),
        lastScrapeStatus: 'success',
        consecutiveFailures: 0,
        updatedAt: new Date(),
      })
      .onConflictDoUpdate({
        target: scraperHealth.retailerId,
        set: {
          lastScrapedAt: new Date(),
          lastSuccessfulScrapeAt: new Date(),
          lastScrapeStatus: 'success',
          consecutiveFailures: 0,
          updatedAt: new Date(),
        },
      });
  });

  events.on('failed', async ({ jobId, failedReason }) => {
    const job = await scrapeQueue.getJob(jobId);
    if (!job) return;
    const { retailerId } = job.data as { retailerId: string };
    const status = failedReason?.toLowerCase().includes('bot block') ? 'blocked' : 'failed';
    await db
      .insert(scraperHealth)
      .values({
        retailerId,
        lastScrapedAt: new Date(),
        lastScrapeStatus: status,
        consecutiveFailures: 1,
        updatedAt: new Date(),
      })
      .onConflictDoUpdate({
        target: scraperHealth.retailerId,
        set: {
          lastScrapedAt: new Date(),
          lastScrapeStatus: status,
          consecutiveFailures: sql`${scraperHealth.consecutiveFailures} + 1`,
          updatedAt: new Date(),
        },
      });
  });

  return events;
}
