```ts
import type { RawProductScrape } from '../adapters/types';
import type { NormalizedProduct } from '@whisky-hunter/shared';

// ─── Numeral word tables (Pitfall 3: normalise before comparison) ───────────
const WORD_TO_NUM: Record<string, number> = {
  one: 1, two: 2, three: 3, four: 4, five: 5, six: 6, seven: 7, eight: 8,
  nine: 9, ten: 10, eleven: 11, twelve: 12, thirteen: 13, fourteen: 14,
  fifteen: 15, sixteen: 16, seventeen: 17, eighteen: 18, nineteen: 19,
  twenty: 20, 'twenty-one': 21, 'twenty-two': 22, 'twenty-three': 23,
  'twenty-four': 24, 'twenty-five': 25, 'twenty-six': 26, 'twenty-seven': 27,
  'twenty-eight': 28, 'twenty-nine': 29, thirty: 30, 'thirty-five': 35,
  forty: 40, 'forty-five': 45, fifty: 50, sixty: 60, seventy: 70,
};

const ROMAN_TO_NUM: Record<string, number> = {
  I: 1, II: 2, III: 3, IV: 4, V: 5, VI: 6, VII: 7, VIII: 8, IX: 9, X: 10,
  XI: 11, XII: 12, XIII: 13, XIV: 14, XV: 15, XVI: 16, XVII: 17, XVIII: 18,
  XIX: 19, XX: 20, XXI: 21, XXV: 25, XXX: 30, XL: 40,
};

/**
 * Replace English numeral words and Roman numerals with digits.
 * 'Fifteen' → '15', 'Twenty-Five' → '25', 'XV' → '15'
 * DATA-02 prerequisite: prevents false age-matching failures (Pitfall 3).
 */
export function expandNumeralWords(input: string): string {
  let s = input;
  // Replace compound numeral words (e.g. 'Twenty-Five') before simple ones
  for (const [word, num] of Object.entries(WORD_TO_NUM).sort(
    ([a], [b]) => b.length - a.length,
  )) {
    s = s.replace(new RegExp(word, 'gi'), String(num));
  }
  // Replace standalone Roman numerals (whole-word match, bounded by space/start/end/punctuation)
  for (const [roman, num] of Object.entries(ROMAN_TO_NUM).sort(
    ([a], [b]) => b.length - a.length,
  )) {
    s = s.replace(new RegExp(`(?<![A-Z])${roman}(?![A-Z])`, 'g'), String(num));
  }
  return s;
}

/** Extract integer age from a product name (after numeral expansion). Returns null for NAS. */
export function extractAge(name: string): number | null {
  const expanded = expandNumeralWords(name);
  // Match patterns like "15 Year Old", "15yo", "15-Year-Old", "15yr", "aged 15"
  const match = expanded.match(/\b(?:aged?\s+)?(\d{1,2})\s*(?:y(?:ear)?s?(?:\s*old)?|yo|-year(?:s)?(?:-old)?)\b/i);
  if (!match) return null;
  const age = parseInt(match[1]!, 10);
  return age >= 1 && age <= 100 ? age : null;
}

/** Extract volume in ml. Handles '70cl', '700ml', '1L', '1.75L'. Returns 700 as default. */
export function extractVolume(name: string): number {
  const mlMatch = name.match(/(\d+(?:\.\d+)?)\s*ml/i);
  if (mlMatch) return Math.round(parseFloat(mlMatch[1]!) );

  const clMatch = name.match(/(\d+(?:\.\d+)?)\s*cl/i);
  if (clMatch) return Math.round(parseFloat(clMatch[1]!) * 10);

  const lMatch = name.match(/(\d+(?:\.\d+)?)\s*[Ll](?:\s|$)/);
  if (lMatch) return Math.round(parseFloat(lMatch[1]!) * 1000);

  return 700; // Default standard bottle size
}

/** Extract ABV percentage. '46%', '46.0% abv', '40% ABV'. Returns null if not found. */
export function extractAbv(name: string): number | null {
  const match = name.match(/(\d{2,3}(?:\.\d)?)\s*%/);
  if (!match) return null;
  const abv = parseFloat(match[1]!);
  return abv >= 20 && abv <= 96 ? abv : null;
}

/** Infer category from name/description keywords. */
function inferCategory(name: string): NormalizedProduct['category'] {
  const n = name.toLowerCase();
  if (n.includes('bourbon') || n.includes('kentucky') || n.includes('tennessee')) return 'bourbon';
  if (n.includes('rye')) return 'rye';
  if (n.includes('irish')) return 'irish';
  if (n.includes('japanese') || n.includes('japan') || n.includes('yamazaki') ||
      n.includes('nikka') || n.includes('hibiki') || n.includes('hakushu')) return 'japanese';
  if (n.includes('single malt')) return 'scotch_single_malt';
  if (n.includes('blended')) return 'scotch_blended';
  return 'other';
}

/** Infer distillery name from the product name (first 1-2 capitalised words). Heuristic. */
function inferDistillery(name: string): string {
  const words = name.trim().split(/\s+/);
  // Distillery is typically the first word; some are two-word (Highland Park, Caol Ila)
  const twoWordDistilleries = ['highland park', 'caol ila', 'glen garioch', 'ben nevis',
    'old forester', 'jack daniels', 'knob creek', 'makers mark'];
  const lower = name.toLowerCase();
  for (const d of twoWordDistilleries) {
    if (lower.startsWith(d)) return d.split(' ').map(w => w[0]!.toUpperCase() + w.slice(1)).join(' ');
  }
  return words[0] ?? 'Unknown';
}

/** Full normalisation pipeline: expand → extract → classify. */
export function normalize(
  raw: RawProductScrape,
  overrides?: { distillery?: string; region?: string; caskType?: string },
): Omit<NormalizedProduct, 'priceLocal' | 'currency' | 'url' | 'imageUrl' | 'sourceProductId' | 'inStock'> & {
  ageYears: number | null;
  volumeMl: number;
  abv: number | null;
} {
  const expanded = expandNumeralWords(raw.name);
  return {
    name: raw.name.trim(),
    distillery: overrides?.distillery ?? inferDistillery(raw.name),
    ageYears: extractAge(expanded),
    volumeMl: extractVolume(raw.name),
    abv: extractAbv(raw.name),
    category: inferCategory(raw.name),
    region: overrides?.region ?? null,
    caskType: overrides?.caskType ?? null,
  };
}
```

```ts
import { distance } from 'fastest-levenshtein';

// DATA-02: LOCKED thresholds — do NOT lower AUTO_MERGE_THRESHOLD without human review
// Lowering auto-merge risks false positives that corrupt the canonical product registry
export const AUTO_MERGE_THRESHOLD = 0.90;
export const REVIEW_QUEUE_THRESHOLD = 0.70;

export interface ProductForResolver {
  name: string;
  distillery: string;
  ageYears: number | null;
  volumeMl: number;
}

function similarity(a: string, b: string): number {
  const maxLen = Math.max(a.length, b.length);
  if (maxLen === 0) return 1;
  return 1 - distance(a, b) / maxLen;
}

/**
 * Normalise a product name for comparison: lowercase, strip stop words, collapse whitespace.
 * Applied to both candidates before Levenshtein scoring.
 */
export function normalizeName(name: string): string {
  return name
    .toLowerCase()
    .replace(/\b(year|years|old|aged?|single|malt|whisky|whiskey|scotch|year-old|yr|y\.o\.)\b/gi, ' ')
    .replace(/\s+/g, ' ')
    .trim();
}

/**
 * Compute a composite match score between two normalized products.
 * Formula: 0.50 * nameSim + 0.30 * distillerySim + 0.10 * ageSim + 0.10 * volumeSim
 *
 * HARD REJECT: If both ageYears are known and different → return 0 immediately.
 * This prevents "Glenfarclas 15" from merging with "Glenfarclas 25" even if
 * name similarity is high (Pitfall 3).
 *
 * @returns score in [0, 1]. 0 = hard reject. >= 0.90 = auto-merge. 0.70-0.89 = review queue.
 */
export function computeMatchScore(a: ProductForResolver, b: ProductForResolver): number {
  // Hard reject: confirmed different ages → never merge
  if (a.ageYears !== null && b.ageYears !== null && a.ageYears !== b.ageYears) {
    return 0;
  }

  const nameSim       = similarity(normalizeName(a.name), normalizeName(b.name));
  const distillerySim = a.distillery.toLowerCase().trim() === b.distillery.toLowerCase().trim() ? 1 : 0;
  const ageSim        = a.ageYears === b.ageYears ? 1 : 0; // null === null is true
  const volumeSim     = a.volumeMl === b.volumeMl ? 1 : 0;

  return 0.50 * nameSim + 0.30 * distillerySim + 0.10 * ageSim + 0.10 * volumeSim;
}

export type MergeDecision =
  | { action: 'auto_merge' }
  | { action: 'review_queue'; score: number }
  | { action: 'no_action' };

/** Classify a match score into an actionable routing decision. */
export function routeMatchDecision(score: number): MergeDecision {
  if (score >= AUTO_MERGE_THRESHOLD) return { action: 'auto_merge' };
  if (score >= REVIEW_QUEUE_THRESHOLD) return { action: 'review_queue', score };
  return { action: 'no_action' };
}
```

```ts
// DATA-05: Staleness detection utility.
// CRITICAL (Pitfall 5): Always use last_successful_scrape_at, NOT last_scraped_at.
// last_scraped_at is updated even on bot-block failures, making it useless for freshness.

/** 48-hour staleness threshold. Listings older than this are considered stale. */
export const STALE_THRESHOLD_HOURS = 48;

/**
 * Returns true if the given timestamp is older than STALE_THRESHOLD_HOURS, or is null
 * (null means the retailer has never successfully scraped — treat as stale).
 *
 * @param lastSuccessfulScrapeAt - Value from scraper_health.last_successful_scrape_at.
 *                                 MUST be last_successful_scrape_at, not last_scraped_at.
 */
export function isStale(lastSuccessfulScrapeAt: Date | null): boolean {
  if (lastSuccessfulScrapeAt === null) return true; // Never successfully scraped
  const ageMs = Date.now() - lastSuccessfulScrapeAt.getTime();
  const thresholdMs = STALE_THRESHOLD_HOURS * 60 * 60 * 1000;
  return ageMs >= thresholdMs;
}
```
